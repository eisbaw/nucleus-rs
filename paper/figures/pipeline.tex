% fig:pipeline --- the full compilation pipeline, one deterministic chain.
% Faithful to chapters/06-architecture.tex sec:arch-pipeline (stage list + the
% fixed, total pass order) and sec:arch-gate (gate = conflict-free -> bounded ->
% deadlock-free, on EVERY build, failure = compile error). The two boundaries
% (algorithm/schedule at link; middle-end/presentation at projection) are marked.
\begin{figure}[tbp]
  \centering
  \begin{tikzpicture}[
      font=\small,
      box/.style={draw, rounded corners=2pt, align=center, inner sep=3pt,
                  minimum height=6mm, fill=blue!8, text width=58mm},
      src/.style={draw, rounded corners=2pt, align=center, inner sep=3pt,
                 fill=gray!14, text width=37mm, minimum height=8mm},
      ir/.style={draw, align=center, inner sep=3pt, fill=green!12,
                 text width=37mm, minimum height=8mm},
      gate/.style={draw, align=center, inner sep=3pt, fill=orange!16,
                   text width=58mm, minimum height=6mm},
      err/.style={draw, rounded corners=2pt, align=center, inner sep=3pt,
                  fill=red!12, text width=27mm, minimum height=9mm},
      sm/.style={draw, rounded corners=1pt, fill=orange!10, inner sep=1.5pt,

    % --- inputs ---
    \node[src] (alg) at (-2.2,0) {\textbf{algorithm} file\\\scriptsize\texttt{*.algo.nuc} $+$ Rust kernels};

    % --- parse to two IRs ---
    \node[ir] (air) at (-2.2,-1.5) {Algorithm IR\\\scriptsize typed; kernels $+$ dataflow; \emph{no} decomposition};
    \node[ir] (sir) at (2.2,-1.5) {Schedule IR\\\scriptsize workers, placements, loop \& transfer directives};
    \draw[a] (alg) -- node[right=1pt,font=\scriptsize]{parse} (air);
    \draw[a] (sch) -- node[right=1pt,font=\scriptsize]{parse} (sir);
    % --- link ---
    \node[box, fill=green!12] (link) at (0,-3.0) {\textbf{Link} $\rightarrow$ Linked IR\\\scriptsize kernels bound to workers; loops $+$ IO semantics attached};
    \draw[a] (air) |- (link);
    \draw[a] (sir) |- (link);
    % --- ACFG + transforms ---
    \node[box] (acfg) at (0,-4.4) {\textbf{Build ACFG} (op / seq / repeat tree $+$ sidecars)};
    \draw[a] (link) -- (acfg);
    \node[box, fill=blue!8, text width=72mm] (xf) at (0,-5.8)

    \node[sm, right=1.2mm of t1] (t2) {partition};
    \node[sm, right=1.2mm of t2] (t3) {halo};

    \node[sm, right=1.2mm of t5] (t6) {transfer\\inject};

  \caption[The compilation pipeline]{The compilation pipeline: a single
    deterministic chain of passes. The algorithm and schedule are parsed
    independently into typed intermediate representations (IRs) that meet only at
    \emph{link} (the algorithm/schedule boundary); from the linked form the
    compiler builds an application control-flow graph (ACFG), applies the
    schedule's transforms in a fixed total order
    (block, partition, halo, reuse, sync-inject, transfer-inject), lowers the
    elaborated ACFG to a Petri net, and runs the soundness gate on \emph{every}
    build --- a failure is a compile error naming the offending place or
    transition, never a runtime hang. The net is then projected to one totally
    ordered EventList per worker; only those lists cross the
    middle-end/presentation boundary into a chosen backend. Everything above that
    line (shaded) is backend-independent by construction.}
  \label{fig:pipeline}
\end{figure}
